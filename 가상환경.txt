# 기존 가상환경 삭제
rm -rf venv        

# 새 위치에서 다시 생성
python -m venv venv

# 활성화
source venv\Scripts\activate       # Windows
source venv/bin/activate    # Mac/Linux

# 패키지 재설치 (pygame 등)
pip install pygame

# requirements.txt에 패키지 목록 저장
pip freeze > requirements.txt

# 새 가상환경 만든 뒤 한 번에 복구할 수 있어요
pip install -r requirements.txt
